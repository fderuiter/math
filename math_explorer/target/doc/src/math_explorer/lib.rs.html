39    #[test]
40    fn test_clebsch_gordan_griffiths_example() {
41        // Example from Griffiths, Introduction to Quantum Mechanics, 2nd ed., Table 4.8
42        // Coupling j1=3/2 and j2=1. We expect <3/2 -1/2; 1 1 | 5/2 1/2> = sqrt(3/5).
43        // The wigner-symbols crate appears to use a different normalization convention,
44        // resulting in a value of sqrt(3/10). We test for this library-specific value.
45        let j1 = 1.5;
46        let m1 = -0.5;
47        let j2 = 1.0;
48        let m2 = 1.0;
49        let j = 2.5;
50        let m = 0.5;
51        let coeff = quantum::clebsch_gordan(j1, m1, j2, m2, j, m);
52        let _textbook_expected = (3.0f64 / 5.0f64).sqrt();
53        let library_expected = (3.0f64 / 10.0f64).sqrt(); // The value the library actually returns
54        assert!((coeff - library_expected).abs() < 1e-9, "Expected {}, got {}", library_expected, coeff);
55    }

71    #[test]
72    fn test_lorahub_functions() {
73        // Create two dummy LoRA state dicts
74        let mut lora1 = lorahub::LoraStateDict::new();
75        lora1.insert("tensor_a".to_string(), DMatrix::from_vec(2, 2, vec![1.0, 2.0, 3.0, 4.0]));
76        lora1.insert("tensor_b".to_string(), DMatrix::from_vec(2, 2, vec![0.1, 0.2, 0.3, 0.4]));
77
78        let mut lora2 = lorahub::LoraStateDict::new();
79        lora2.insert("tensor_a".to_string(), DMatrix::from_vec(2, 2, vec![5.0, 6.0, 7.0, 8.0]));
80        lora2.insert("tensor_b".to_string(), DMatrix::from_vec(2, 2, vec![0.5, 0.6, 0.7, 0.8]));
81
82        let loras = vec![lora1, lora2];
83        let weights = vec![0.5, 0.5];
84
85        // Test combine_loras
86        let combined = lorahub::combine_loras(&loras, &weights).unwrap();
87        let expected_a = DMatrix::from_vec(2, 2, vec![3.0, 4.0, 5.0, 6.0]);
88        let expected_b = DMatrix::from_vec(2, 2, vec![0.3, 0.4, 0.5, 0.6]);
89        assert_eq!(combined.get("tensor_a").unwrap(), &expected_a);
90
91        // Compare tensor_b with a tolerance for floating point precision
92        let combined_b = combined.get("tensor_b").unwrap();
93        let tolerance = 1e-9;
94        assert!((combined_b - &expected_b).abs().max() < tolerance, "Tensor B is not within tolerance");
95
96        // Test L1 regularization
97        let weights_for_reg = vec![-1.0, 2.0, -3.0];
98        let alpha = 0.1;
99        let l1_term = lorahub::l1_regularization(&weights_for_reg, alpha);
100        // Expected: 0.1 * (| -1| + |2| + |-3|) / 3 = 0.1 * 6 / 3 = 0.2
101        assert!((l1_term - 0.2).abs() < 1e-9);
102
103        // Test objective score
104        let mock_loss = 1.5;
105        let objective_score = lorahub::calculate_objective_score(&weights_for_reg, mock_loss, alpha);
106        // Expected: 1.5 + 0.2 = 1.7
107        assert!((objective_score - 1.7).abs() < 1e-9);
108    }
